class Audited(models.Model):
    """An abstract base: emits NO table of its own, and its two columns merge into every child.

    Its `Meta` carries `unique_together`, which Django installs on a child that declares no Meta
    at all — and only on such a child. Both halves are asserted below.
    """

    created_on = models.DateTimeField(auto_now_add=True)
    source_tag = models.CharField(max_length=8, default="base")

    class Meta:
        abstract = True
        unique_together = ("created_on", "source_tag")
        # Django DOES inherit db_table onto a child that declares no Meta — which would point every
        # child of this base at one table. Refused deliberately, and reported. Present here so the
        # "not inherited" assertion has something that could fail.
        db_table = "aud_base"


class Enrolment(Audited):
    """Concrete child. `source_tag` is REDECLARED here, so the child's max_length=32 must win over the
    base's max_length=8 — that is what proves the merge order rather than merely the merge.

    It declares its own `Meta`, so Django does NOT install `Audited`'s: this model must come out
    with no `unique_together` constraint at all.
    """

    serial = models.CharField(max_length=20)
    source_tag = models.CharField(max_length=32, default="enrolment")

    class Meta:
        db_table = "f1_enrolment_legacy"
        ordering = ["serial"]


class Occurrence(Audited):
    """Concrete child with NO Meta of its own — so Django installs the abstract base's, and
    `unique_together` reaches this table while it does not reach Enrolment's."""

    description = models.CharField(max_length=100)

class Engineer(models.Model):
    licence_no = models.CharField(max_length=11)
    crew_ref = models.ForeignKey(Crew, on_delete=models.CASCADE)
    enabled = models.BooleanField(default=True)
    nickname = models.CharField(max_length=30)

    class Meta:
        constraints = [
            # IMPORTED. `crew_ref` resolves to the imported `crew_ref_id` column.
            models.UniqueConstraint(fields=["licence_no", "crew_ref"], name="uniq_engineer_licence_crew"),
            # IMPORTED (#934) — a PARTIAL unique index, its Q() through the CheckConstraint
            # translator: condition = "\"enabled\"".
            models.UniqueConstraint(
                fields=["nickname"], condition=Q(enabled=True), name="uniq_nickname_enabled"
            ),
            # IMPORTED (#934) — a FUNCTIONAL unique index, in Django's own SQL spelling.
            models.UniqueConstraint(Lower("nickname"), name="uniq_nickname_lower"),
            # REJECTED — a condition outside the translated set. Imported unconditionally it would
            # start rejecting rows the live database accepts, which is silent data loss on write.
            models.UniqueConstraint(
                fields=["licence_no"], condition=Q(nickname__startswith="A"), name="uniq_licence_nickname_a"
            ),
            # REJECTED — `deferrable=` moves the check to commit time, which PormG cannot declare.
            models.UniqueConstraint(fields=["licence_no"], deferrable=models.Deferrable.DEFERRED, name="uniq_licence_deferred"),
            # REJECTED — no PormG equivalent at all.
            models.CheckConstraint(check=Q(licence_no__regex=r"^\d{11}$"), name="chk_licence_digits"),
        ]
        indexes = [
            # TRANSLATED to `db_index = true` on the field. A one-column CREATE INDEX is exactly
            # what `db_index` emits, and PormG's Index is the COMPOSITE primitive.
            models.Index(fields=["nickname"]),
            # IMPORTED as Models.Index. `crew_ref` resolves to `crew_ref_id`, as on constraints.
            models.Index(fields=["licence_no", "crew_ref"], name="idx_engineer_licence_crew"),
            # IMPORTED (#29) with the direction kept: Models.Index(fields = ("-licence_no", "nickname")).
            models.Index(fields=["-licence_no", "nickname"], name="idx_engineer_licence_desc"),
            # IMPORTED (#29 part 2) as Models.Index(expressions = ("LOWER(\"nickname\")",)) — the
            # functional index in Django's own spelling. (Rejected before #29 part 2.)
            models.Index(Lower("nickname"), name="idx_nickname_lower"),
            # IMPORTED (#29) as method = "gin" — a PostgreSQL-only index, which SQLite refuses.
            GinIndex(fields=["licence_no", "nickname"], name="gin_engineer"),
            # IMPORTED (#29 part 2) as a partial index: condition = "\"enabled\"" (Django's own spelling, #934), through the
            # CheckConstraint translator. (Rejected before #29 part 2.)
            models.Index(fields=["licence_no", "enabled"], condition=Q(enabled=True), name="idx_enabled"),
        ]
        index_together = (("nickname", "enabled"),)


class Penalty(models.Model):
    """#768: CheckConstraints whose Q(...) translates to SQL, over the PHYSICAL columns — `cap`
    is `cap_days` in the table and `engineer` is `engineer_id`."""

    engineer = models.ForeignKey(Engineer, on_delete=models.CASCADE, null=True)
    days = models.IntegerField()
    cap = models.IntegerField(db_column="cap_days")
    variety = models.CharField(max_length=20)
    approved = models.BooleanField(default=False)
    start_date = models.DateField()
    end_date = models.DateField()

    class Meta:
        constraints = [
            # TRANSLATED — `&` between two Q objects, and an F() through `db_column`.
            models.CheckConstraint(check=Q(days__gte=0) & Q(days__lte=F("cap")), name="ck_pen_days"),
            # TRANSLATED — Django 5.1's `condition=` spelling; `in` over strings, one with a quote.
            models.CheckConstraint(
                condition=Q(variety__in=["caution", "reprimand", "driver's choice"]) | Q(approved=False),
                name="ck_pen_kind",
            ),
            # TRANSLATED — `~`, and an FK field resolved to its `_id` column.
            models.CheckConstraint(check=~Q(engineer__isnull=True) | Q(approved=False), name="ck_pen_engineer"),
            # TRANSLATED — `range` becomes BETWEEN.
            models.CheckConstraint(check=Q(cap__range=(1, 365)), name="ck_pen_cap"),
            # STAYS A STUB — two DateFields: SQLite stores dates in more than one representation, so
            # the comparison would not mean the same thing on both engines.
            models.CheckConstraint(check=Q(end_date__gte=F("start_date")), name="ck_pen_dates"),
        ]

class EngineerSerializer(serializers.ModelSerializer):
    """Unlike `RaceReportForm` above, a serializer DOES declare field-shaped members — so the field
    gate alone would import it as a table. `Meta.model` is what keeps it out: it is the signature
    of a class that describes a model, and Django's own ModelBase rejects the attribute on a real
    model ("class Meta got invalid attribute(s): model")."""

    nickname = serializers.CharField(max_length=30)

    class Meta:
        model = Engineer
        fields = ["nickname"]


class EngineerAdminForm(forms.ModelForm):

    must stay silent; this shape must not."""

    def enabled_ones(self):
        return self

class BrokenKey(models.Model):
