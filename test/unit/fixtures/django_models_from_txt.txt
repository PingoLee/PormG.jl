from django.contrib.auth.models import AbstractUser
from django.db import models
from django.db.models.deletion import RESTRICT

class Timing_server(models.Model):
    full_name = models.CharField(max_length=250)
    host = models.CharField(max_length=250)
    port = models.CharField(max_length=250)
    user = models.CharField(max_length=250)
    password = models.CharField(max_length=250)

class Race_venue(models.Model):
    full_name = models.CharField(max_length=250, blank=True, null=True)
    enabled = models.BooleanField(default=True)
    refreshed_on = models.DateField(null=True, blank=True)
    venue_code = models.BigIntegerField(null=True)
    server = models.ForeignKey(Timing_server, on_delete=models.RESTRICT, null=True, db_constraint=False)
    statement_no = models.IntegerField(null=True)

class Grand_PRIX(models.Model):
    venue = models.ForeignKey(Race_venue, on_delete=models.CASCADE, db_constraint=False)
    gp_label = models.CharField(max_length=250, blank=True, null=True)
    notes = models.TextField(null=True)
    host_gp = models.ForeignKey("Grand_PRIX", on_delete=models.SET_NULL, null=True, related_name="hosted_gps", db_constraint=False)
    online_booking = models.CharField(default="0", choices=(("0", "Not deployed"), ("1", "Deploying"), ("2", "Deployed")), max_length=1)
    medical_def = models.JSONField(null=True, blank=True)

class TeamPrincipal(AbstractUser):
    user_type = models.CharField(default=3, choices=((1, "HOD"), (2, "Consultant"), (3, "Principal")), max_length=10)
    venue_ref = models.ForeignKey(Race_venue, on_delete=models.RESTRICT, default=0, null=False, blank=False, db_constraint=False)
    home_gp = models.ForeignKey(Grand_PRIX, on_delete=RESTRICT, default=0, null=False, blank=False, db_constraint=False)
    refreshed_on = models.DateField(null=True, blank=True)
    viewer = models.BooleanField(default=False)

class LoginToken(models.Model):
    user = models.ForeignKey(TeamPrincipal, on_delete=models.CASCADE, related_name="login_tokens")
    token = models.CharField(max_length=64, unique=True, db_index=True)
    created_at = models.DateTimeField(auto_now_add=True)
    expires_at = models.DateTimeField()
    used = models.BooleanField(default=False)

class IssueReport(models.Model):
    user = models.ForeignKey(TeamPrincipal, on_delete=models.CASCADE, related_name="issue_reports")
    venue = models.ForeignKey(Race_venue, on_delete=models.RESTRICT, blank=True, null=True, related_name="issue_reports")
    variety = models.CharField(max_length=20, choices=(("BUG", "Bug"), ("SUGGESTION", "Suggestion")), default="BUG")
    status = models.CharField(max_length=20, choices=(("PENDING", "Pending"), ("RESOLVED", "Resolved")), default="PENDING")
    description = models.TextField()
    image = models.FileField(upload_to="reports/", null=True, blank=True)
    reply = models.TextField(blank=True, default="")
    reply_at = models.DateTimeField(null=True, blank=True)
    user_read_at = models.DateTimeField(null=True, blank=True)
    reply_by = models.ForeignKey("TeamPrincipal", on_delete=models.SET_NULL, null=True, blank=True, related_name="reports_answered")
    opened_on = models.DateTimeField(auto_now_add=True)

class Pit_crewHQ(models.Model):
    id = models.AutoField(primary_key=True)
    user = models.OneToOneField(TeamPrincipal, on_delete=models.CASCADE, db_constraint=False)
    created_on = models.DateTimeField(auto_now=True)
    objects = models.Manager()

class Lap_telemetry(models.Model):
    venue = models.ForeignKey(Race_venue, on_delete=models.CASCADE)
    weight_kg = models.FloatField(null=True)
    height_cm = models.FloatField(null=True)
    slot_time = models.TimeField(null=True)
    service_time = models.DurationField(null=True)

class Grand_prix_group(models.Model):
    venue = models.ForeignKey(Race_venue, on_delete=models.CASCADE, db_constraint=False, related_name="gp_groups")
    full_name = models.CharField(max_length=120)
    is_public = models.BooleanField(default=False)
    authored_by = models.ForeignKey("TeamPrincipal", on_delete=models.SET_NULL, null=True, blank=True, db_constraint=False, related_name="gp_groups")
    gps = models.ManyToManyField(Grand_PRIX, blank=True, related_name="gp_groups")
    created_on = models.DateTimeField(auto_now_add=True)
    updated_on = models.DateTimeField(auto_now=True)

class Media_programme(models.Model):
    programme_code = models.CharField(max_length=50, unique=True, db_index=True)
    full_name = models.CharField(max_length=150)
    enabled = models.BooleanField(default=True)
    target_tier = models.PositiveSmallIntegerField(default=3)
    allowed_kinds = models.JSONField(default=list, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
